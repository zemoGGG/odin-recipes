ingredient_groups:
  - items:
      - 1 tbsp olive oil
      - 1 onion, diced
      - 2 lbs ground beef
      - 3 cloves garlic, minced
      - 2 tbsp chili powder
      - 1 (28 oz) can crushed tomatoes
      - 2 (15 oz) cans kidney beans, drained
step_groups:
  - name: Brown
    steps:
      - Heat the oil in a large pot over medium-high heat. Add the onion and cook until soft, about 5 minutes.
      - Add the beef and cook, breaking it up, until browned. Stir in the garlic and chili powder for 1 minute.
  - name: Simmer
    steps:
      - Add the tomatoes and beans. Bring to a simmer, then cook uncovered for 30–40 minutes, stirring now and then.
      - Season with salt and pepper and serve.
