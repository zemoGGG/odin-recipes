ingredient_groups:
  - name: Spice mix
    items:
      - Paprika (2 tbsp)
      - Black pepper (2 tbsp, freshly ground)
      - Garlic powder (2 tsp)
      - Oregano (2 tsp, dried)
      - Cayenne (1/2 tsp)
  - name: Buttermilk brine
    items:
      - Buttermilk (1 cup)
      - Egg (1)
      - Salt
  - name: Chicken
    items:
      - Chicken breasts/thighs/drumsticks/wings (2–3 lbs)
      - Flour (1 1/2 cups)
      - Cornstarch (1/2 cup)
      - Baking powder (1 tsp)
      - Vegetable oil/shortening (3–4 cups)
step_groups:
  - steps:
      - Mix paprika, black pepper, garlic powder, oregano, and cayenne in a small bowl.
      - Whisk buttermilk, egg, 1 tbsp salt, and 2 tbsp spice mixture in a large bowl. Add the chicken and coat evenly. Transfer to a ziploc bag and marinate in the fridge from 4 hours to overnight.
      - Whisk flour, cornstarch, baking powder, 2 tsp salt, and the remaining spice mixture in a bowl. Add 3 tbsp of the marinade to the flour and mix together. Coat all chicken pieces.
      - Heat oil in a dutch oven/cast iron to 375°F. Fry chicken pieces, keeping the oil temperature between 325°F and 350°F.
      - Transfer to a wire rack, and season lightly with kosher salt. Serve immediately.
